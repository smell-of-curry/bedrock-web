// A conservative MTU keeps resource-pack datagrams below common tunnel limits.
// Returning the original RakNet connection preserves its packet capabilities.
type webRakNet struct{ logger *slog.Logger }

func (n webRakNet) DialContext(ctx context.Context, address string) (net.Conn, error) {
	return (raknet.Dialer{ErrorLog: n.logger, MaxMTU: 1200}).DialContext(ctx, address)
}
func (n webRakNet) PingContext(ctx context.Context, address string) ([]byte, error) {
	return (raknet.Dialer{ErrorLog: n.logger, MaxMTU: 1200}).PingContext(ctx, address)
}
func (n webRakNet) Listen(address string) (minecraft.NetworkListener, error) {
	return (raknet.ListenConfig{ErrorLog: n.logger}).Listen(address)
}
func init() {
	minecraft.RegisterNetwork("raknet", func(logger *slog.Logger) minecraft.Network { return webRakNet{logger: logger} })
}

// Tokens travel only over inherited private pipes to the personal relay. They
// are never printed, persisted, included in packet traces, or sent to browsers.
type webTokenSource struct {
	source oauth2.TokenSource
	output *os.File
	mu     sync.Mutex
}

func (s *webTokenSource) Token() (*oauth2.Token, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	token, err := s.source.Token()
	if err != nil {
		return nil, err
	}
	if err := json.NewEncoder(s.output).Encode(token); err != nil {
		return nil, fmt.Errorf("save session to relay: %w", err)
	}
	return token, nil
}

// The public PokéBedrock hub uses Dragonfly, which accepts click-air/block in
// InventoryTransaction but rejects those duplicates in PlayerAuthInput. BDS
// needs the latter, so this compatibility rule is scoped to the verified hub.
type webProtocol struct{ minecraft.Protocol }

func (p webProtocol) ConvertFromLatest(pk packet.Packet, conn *minecraft.Conn) []packet.Packet {
	if input, ok := pk.(*packet.PlayerAuthInput); ok {
		if action, present := input.ItemInteractionData.Value(); present && action.ActionType != protocol.UseItemActionBreakBlock {
			copy := *input
			copy.InputData = protocol.NewInputFlags(input.InputData.Len())
			for i := 0; i < input.InputData.Len(); i++ {
				if input.InputData.Load(i) {
					copy.InputData.Set(i)
				}
			}
			copy.InputData.Unset(packet.InputFlagPerformItemInteraction)
			copy.ItemInteractionData = protocol.Optional[protocol.UseItemTransactionData]{}
			return p.Protocol.ConvertFromLatest(&copy, conn)
		}
	}
	return p.Protocol.ConvertFromLatest(pk, conn)
}
func configureWebAuthentication(ctx context.Context, dialer *minecraft.Dialer, address string) error {
	host, port, _ := net.SplitHostPort(address)
	if strings.EqualFold(strings.TrimSuffix(host, "."), "play.pokebedrock.com") && port == "19132" {
		dialer.Protocol = webProtocol{minecraft.DefaultProtocol}
	}
	// Record packet identifiers once per direction. No packet payload is read.
	seen := map[string]bool{}
	var traceMu sync.Mutex
	dialer.PacketFunc = func(h packet.Header, _ []byte, src, dst net.Addr) {
		traceMu.Lock()
		defer traceMu.Unlock()
		marker := fmt.Sprintf("id=%d src=%s dst=%s", h.PacketID, src.String(), dst.String())
		if !seen[marker] {
			seen[marker] = true
			fmt.Fprintln(os.Stdout, "[BEDROCK_PACKET]", marker)
		}
	}
	dialer.ClientData.DeviceModel = "Bedrock Web"
	if os.Getenv("BEDROCK_WEB_AUTH") != "xbox" {
		return nil
	}
	var cached *oauth2.Token
	if err := json.NewDecoder(io.LimitReader(os.Stdin, 65536)).Decode(&cached); err != nil && err != io.EOF {
		return fmt.Errorf("read relay session: %w", err)
	}
	var token *oauth2.Token
	if cached != nil {
		token, _ = auth.RefreshTokenSourceWriter(cached, io.Discard).Token()
	}
	if token == nil {
		var err error
		token, err = auth.RequestLiveTokenContext(ctx, os.Stdout)
		if err != nil {
			return err
		}
	} else {
		fmt.Fprintln(os.Stdout, "Authentication successful.")
	}
	output := os.NewFile(3, "relay-session")
	dialer.TokenSource = &webTokenSource{source: auth.RefreshTokenSourceWriter(token, io.Discard), output: output}
	return nil
}
